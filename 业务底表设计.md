## 报表信息底表

这个表要经过数据中台/ETL处理：  
来源表为finedb里面的fine_authority_object

批次:t+1

```sql
create table if not exists dim_bi_report_info_dtl (
    report_id string comment "报表ID",
    report_name string comment "报表名称",
    report_type string comment "报表类型BI/FRM/FVS/CPT",
    report_device_type string comment "支持查阅的设备类型",
    report_expandtype string comment "报表扩展类型",
    report_develop_path string comment "报表的开发路径",
    report_pure_develop_path string comment "优化后报表开发路径",
    report_displayname_path_ids Array<string> comment "报表前端路径ID集合",
    report_displayname_path string comment "报表的前端查看路径",
    etl_create_time string COMMENT '首次入仓时间',
    etl_update_time string COMMENT '最近入仓时间'
)comment'报表访问日志记录表'
stored as orc 
*/


insert overwrite table dim_bi_report_info_dtl
  SELECT
      distinct
      target_id as report_id,                                                   -- 报表ID
      p.displayname as report_name,                                             -- 报表名称
      object_type as report_type,                                               -- 报表类型
      report_device_type as report_device_type,                                 -- 报表查阅设备 
      expandType as report_expandtype,                                          -- 报表扩展类型
      path as report_develop_path,                                              -- 报表开发路径
      REGEXP_REPLACE(path, '^([0-9]+)[^/]*/', '$1/')  as report_pure_develop_path, -- 优化后报表开发路径
      collect_list(f.id) as report_displayname_path_ids,
      concat_ws('/',collect_list(f.displayname)) AS report_displayname_path,    -- 报表前端路径
    '${bdp.system.curtime}' as etl_create_time,                                 -- 首次入仓时间
    '${bdp.system.curtime}' as etl_update_time                                  -- 本次入仓时间  
  FROM
      (   
        SELECT
            id AS target_id,
            displayname,
            path,
            expandType,
            case 
                when path like "%frm%" then 'frm'
                when path like "%fvs" then 'fvs'
                when path like "%cpt%" then 'cpt'
                when path like "%https%" then 'url'
                else 'bi'
            end as object_type,
            case 
                when deviceType in (1,3) then 'PC'
                when deviceType in (4,6) then 'Phone'
                when deviceType in (0,5,7) then 'both'
                else 'Pad'
            end as report_device_type,
            EXPLODE(split(fullpath,
          '-_-')) AS path_id        
      FROM
          ods_fine_authority_object        
      WHERE
          pt = "${bdp.system.bizdate}"           
           ) p  
  left JOIN
      (select
            id,
            displayname
      from
          ods_fine_authority_object      
      where
          pt = "${bdp.system.bizdate}" ) f          
          ON p.path_id = f.id  
  where
      path_id not in ('decision-directory-root','')  
  GROUP BY
      p.target_id,
      p.path,
      p.expandType,
      p.object_type,
      p.report_device_type,
      p.displayname
```

## 报表访问日志

```sql
create table if not exists dim_bi_visit_log (
    day_id string comment "日期ID",
    report_id string comment "报表ID",
    report_desc string comment "报表名称",
    report_type string comment "报表类型",
    report_device_type string comment "报表所属类型(PC/手机)",
    report_design_path string comment "开发路径",
    report_frontend_path string comment "目录路径",
    user_code string comment "用户登录账号",
    user_name string comment "用户登陆名称",
    user_roles string comment "用户当前角色",
    visit_type string comment "访问类型",
    visit_num string comment "访问次数"
)comment'报表访问日志记录表'partitioned by(
    pt string
 )stored as orc 
 lifecycle 3;
*/
insert overwrite table dim_bi_visit_log_inc PARTITION (pt = "${bdp.system.bizdate}")
select 
  fre.day_id,
  ct.report_id,
  ifnull(ct.report_name,fre.displayName) as report_name,
  ct.report_type,
  ct.report_device_type,
  ifnull(ct.report_develop_path,fre.tname) as report_develop_path,
  ct.report_displayname_path report_displayname_path,
  user_code,
  user_name,
  role_name,
  type,
  ifnull(visit_num,0) visit_num
from 
  (
        select 
            date_format(time,'yyyy-MM-dd') day_id,
            type,
            tname,
            regexp_replace(tname, '^(\\d+)[^/]*/', '$1/') tname_trans,
            displayName,
            source,
            regexp_extract(username, '\\((\\d+)\\)', 1) AS user_code,
            regexp_extract(username, '^(.*?)\\(', 1) AS user_name,
            count(id) as visit_num
        from 
            (select * from ods_fine_record_execute where pt="${bdp.system.bizdate}" 
            and displayName is not null 
            and date_format(time,'yyyyMMdd') = "${bdp.system.bizdate}" ) logs
        group by 
            day_id,
            type,
            tname,
            source,
            tname_trans,
            displayName,
            username
  ) fre 
  left join (
    select 
      *,concat(report_displayname_path,'/',report_name) as report_full_path 
    from dim_bi_report_info_dtl
  ) ct 
    on (case when source is null then fre.tname_trans = ct.report_pure_develop_path else fre.displayName =  ct.report_full_path end)
  left join dim_bi_user user_info on user_info.userName=fre.user_code
```
